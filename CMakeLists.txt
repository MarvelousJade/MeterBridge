cmake_minimum_required(VERSION 3.24)
project(MeterBridge VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(METERBRIDGE_STORAGE_ONLY "Build portable reading/storage tests, not the gateway" OFF)
option(METERBRIDGE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(meter-storage src/reading.cpp src/store.cpp)
target_include_directories(meter-storage PUBLIC src)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(meter-storage PUBLIC SQLite3::SQLite3 Threads::Threads)
else() # CMake 3.24 compatibility
  target_link_libraries(meter-storage PUBLIC SQLite::SQLite3 Threads::Threads)
endif()
if(MSVC)
  target_compile_options(meter-storage PUBLIC /W4 /WX)
else()
  target_compile_options(meter-storage PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
if(METERBRIDGE_SANITIZE)
  target_compile_options(meter-storage PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(meter-storage PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(storage-tests tests/storage.cpp)
  target_link_libraries(storage-tests PRIVATE meter-storage)
  foreach(scenario baseline commit_failure partial_prune invalid_capacity)
    add_test(NAME storage_${scenario} COMMAND storage-tests ${scenario})
  endforeach()
endif()
if(METERBRIDGE_STORAGE_ONLY)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "MeterBridge targets Linux; use WSL on Windows.")
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(MODBUS REQUIRED IMPORTED_TARGET libmodbus)
pkg_check_modules(MQTT REQUIRED IMPORTED_TARGET libmosquitto)

include(FetchContent)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib
  URL https://codeload.github.com/yhirose/cpp-httplib/tar.gz/refs/tags/v0.18.3
  URL_HASH SHA256=a0567bcd6c3fe5cef1b329b96245119047f876b49e06cc129a36a7a8dffe173e
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(httplib)

add_library(meter-core src/core.cpp)
target_include_directories(meter-core PUBLIC src)
target_link_libraries(meter-core PUBLIC PkgConfig::MODBUS meter-storage)
target_compile_options(meter-core PUBLIC -Wall -Wextra -Wpedantic -Werror)
add_executable(meterbridge src/main.cpp src/mqtt.cpp)
target_link_libraries(meterbridge PRIVATE meter-core PkgConfig::MQTT httplib::httplib)
add_executable(meter-sim src/simulator.cpp)
target_link_libraries(meter-sim PRIVATE meter-core)

if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE meter-core)
  add_test(NAME core COMMAND core-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE_DIR:meterbridge>)
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
endif()
install(TARGETS meterbridge meter-sim RUNTIME DESTINATION bin)
